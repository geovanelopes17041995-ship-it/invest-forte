import streamlit as st
import pandas as pd
import random

# Configuração da página
st.set_page_config(
    page_title="Invest Forte | Plataforma Financeira", 
    page_icon="📈",
    layout="centered"
)

# Estilização CSS customizada
st.markdown("""
<style>
    .stApp { background-color: #0B0E14; }
    .card-saldo {
        background: linear-gradient(135deg, #1E2640 0%, #0F172A 100%);
        border: 1px solid #334155;
        border-radius: 16px;
        padding: 20px;
        margin-bottom: 20px;
    }
    .card-indicacao {
        background-color: #161E2E;
        border: 1px solid #283548;
        border-radius: 12px;
        padding: 16px;
        margin-bottom: 16px;
    }
    .stButton>button {
        width: 100%;
        border-radius: 8px;
        font-weight: 600;
        background-color: #0ea5e9;
        color: white;
        border: none;
        padding: 10px 0px;
    }
    .stButton>button:hover { background-color: #0284c7; color: white; }
</style>
""", unsafe_allow_html=True)

# --- INICIALIZAÇÃO DA BASE DE DADOS EM MEMÓRIA ---
if "usuarios" not in st.session_state:
    st.session_state.usuarios = {
        "admin@investforte.com": {
            "nome": "Cliente Demonstração",
            "senha": "123",
            "saldo": 1250.00,
            "extrato": [],
            "codigo_ref": "REF123",
            "indicado_por": None,
            "bono_recebido": False
        }
    }

if "usuario_logado" not in st.session_state:
    st.session_state.usuario_logado = None

if "saques_bloqueados_global" not in st.session_state:
    st.session_state.saques_bloqueados_global = False

# VERIFICAÇÃO SE É O SEU ACESSO MASTER VIA URL (ex: site.com/?admin=true)
query_params = st.query_params
eh_admin_master = query_params.get("admin") == "true"

# Topo da aplicação
col_logo, col_user = st.columns([2, 1])
with col_logo:
    st.markdown("## ⚡ **INVEST FORTE**")
    st.caption("Plataforma Digital de Investimentos")

with col_user:
    if st.session_state.usuario_logado:
        st.write(f"👤 **{st.session_state.usuarios[st.session_state.usuario_logado]['nome']}**")
        if st.button("Sair (Logout)"):
            st.session_state.usuario_logado = None
            st.rerun()

st.divider()

# ---------------------------------------------------------
# TELA DE LOGIN E CADASTRO
# ---------------------------------------------------------
if st.session_state.usuario_logado is None:
    # Se você acessar com ?admin=true na URL, o menu exibe a opção Master
    opcoes_menu = ["🔑 Entrar", "📝 Criar Conta"]
    if eh_admin_master:
        opcoes_menu.append("⚙️ Painel Master")

    aba_acesso = st.radio("", opcoes_menu, horizontal=True, label_visibility="collapsed")
    
    if aba_acesso == "🔑 Entrar":
        st.markdown("### 🔑 Entrar na sua Conta")
        email = st.text_input("E-mail:")
        senha = st.text_input("Senha:", type="password")
        
        if st.button("Aceder à Conta"):
            if email in st.session_state.usuarios and st.session_state.usuarios[email]["senha"] == senha:
                st.session_state.usuario_logado = email
                st.success("Login efetuado com sucesso!")
                st.rerun()
            else:
                st.error("E-mail ou senha incorretos.")

    elif aba_acesso == "📝 Criar Conta":
        st.markdown("### 📝 Criar Nova Conta")
        novo_nome = st.text_input("Nome Completo:")
        novo_email = st.text_input("E-mail de Acesso:")
        nova_senha = st.text_input("Defina uma Senha:", type="password")
        confirma_senha = st.text_input("Confirme a Senha:", type="password")
        codigo_indicacao = st.text_input("Código de Indicação (Opcional):", placeholder="Ex: REF123")
        
        if st.button("Concluir Cadastro"):
            if not novo_nome or not novo_email or not nova_senha:
                st.error("Preencha todos os campos obrigatórios.")
            elif nova_senha != confirma_senha:
                st.error("As senhas não coincidem.")
            elif novo_email in st.session_state.usuarios:
                st.error("Este e-mail já se encontra registado.")
            else:
                quem_indicou = None
                if codigo_indicacao.strip():
                    for u_email, u_dados in st.session_state.usuarios.items():
                        if u_dados.get("codigo_ref") == codigo_indicacao.strip():
                            quem_indicou = u_email
                            break

                novo_ref = f"REF{random.randint(1000, 9999)}"
                st.session_state.usuarios[novo_email] = {
                    "nome": novo_nome,
                    "senha": nova_senha,
                    "saldo": 0.00,
                    "extrato": [],
                    "codigo_ref": novo_ref,
                    "indicado_por": quem_indicou,
                    "bono_recebido": False
                }
                st.success("Conta criada com sucesso! Mude para a aba 'Entrar' para aceder.")

    elif aba_acesso == "⚙️ Painel Master":
        st.markdown("### 🔒 Acesso Administrativo Master")
        senha_master = st.text_input("Senha Master:", type="password")
        if senha_master == "1234":
            st.success("Autenticado como Administrador Master")
            
            st.markdown("---")
            st.markdown("### 🚨 CONTROLE GLOBAL DE SEGURANÇA")
            bloqueio = st.toggle("🔒 BLOQUEAR TODOS OS SAQUES E TRANSFERÊNCIAS", value=st.session_state.saques_bloqueados_global)
            if bloqueio != st.session_state.saques_bloqueados_global:
                st.session_state.saques_bloqueados_global = bloqueio
                st.rerun()
                
            st.markdown("---")
            st.markdown("### 👥 Gestão de Utilizadores")
            for usr_email, dados in st.session_state.usuarios.items():
                st.write(f"**{dados['nome']}** (`{usr_email}`)")
                st.write(f"Saldo Atual: **R$ {dados['saldo']:,.2f}** | Código Ref: `{dados.get('codigo_ref')}`")
                
                c_val, c_btn = st.columns([2, 1])
                val_ajuste = c_val.number_input(f"Ajustar Saldo (R$)", key=f"val_{usr_email}", step=50.0)
                if c_btn.button("Adicionar Saldo", key=f"add_{usr_email}"):
                    dados["saldo"] += val_ajuste
                    dados["extrato"].append({"tipo": "Crédito Admin", "valor": val_ajuste, "status": "Aprovado", "descricao": "Ajuste manual pelo Master"})
                    st.success("Saldo atualizado!")
                    st.rerun()
                st.divider()

# ---------------------------------------------------------
# TELA PRINCIPAL (LOGADO)
# ---------------------------------------------------------
else:
    usr = st.session_state.usuarios[st.session_state.usuario_logado]
    
    aba_ativa = st.radio(
        "", 
        ["📊 Minha Conta", "📥 Depositar", "📤 Saque Pix", "🤝 Indique & Ganhe", "📜 Extrato"], 
        horizontal=True,
        label_visibility="collapsed"
    )

    # ABA 1: DASHBOARD
    if aba_ativa == "📊 Minha Conta":
        st.write("")
        st.markdown(f"""
        <div class="card-saldo">
            <span style="color: #94A3B8; font-size: 14px;">Saldo na Plataforma</span>
            <h1 style="color: #FFFFFF; font-size: 38px; margin: 5px 0px;">R$ {usr['saldo']:,.2f}</h1>
            <span style="color: {'#EF4444' if st.session_state.saques_bloqueados_global else '#10B981'}; font-size: 13px;">
                ● Status dos Saques: {'🚫 SUSPENSOS PELO ADMIN' if st.session_state.saques_bloqueados_global else '✅ LIBERADOS'}
            </span>
        </div>
        """, unsafe_allow_html=True)

        st.markdown("### 💎 Cotas e Oportunidades")
        cotas = [
            {"nome": "Cota Renda Fixa Max", "minimo": 50.0, "retorno": "115% do CDI"},
            {"nome": "Cota FII Imobiliário Prime", "minimo": 100.0, "retorno": "0.9% ao mês em dividendos"}
        ]

        for c in cotas:
            with st.container():
                col_info, col_acao = st.columns([3, 1])
                with col_info:
                    st.markdown(f"**{c['nome']}**")
                    st.caption(f"Aporte Mínimo: R$ {c['minimo']:.2f} • Retorno: {c['retorno']}")
                with col_acao:
                    st.write("")
                    if st.button("Investir", key=c["nome"]):
                        if st.session_state.saques_bloqueados_global:
                            st.error("Operações temporariamente bloqueadas pelo sistema.")
                        elif usr["saldo"] >= c["minimo"]:
                            usr["saldo"] -= c["minimo"]
                            usr["extrato"].append({"tipo": "Investimento", "valor": -c["minimo"], "status": "Aprovado", "descricao": c["nome"]})
                            st.success(f"Aporte realizado na {c['nome']}!")
                            st.rerun()
                        else:
                            st.error("Saldo insuficiente na conta!")
                st.divider()

    # ABA 2: DEPOSITAR VIA PIX
    elif aba_ativa == "📥 Depositar":
        st.markdown("### 📥 Depositar Saldo via Pix")
        valor_deposito = st.number_input("Valor do Depósito (R$):", min_value=10.0, value=200.0, step=10.0)

        if st.button("Gerar QrCode / Pix Copia e Cola"):
            txid = f"TX{random.randint(100000, 999999)}"
            st.success("✅ Código Pix gerado!")
            st.code(f"00020126580014BR.GOV.BCB.PIX0136admin@investforte.com.br5204000053039865405{valor_deposito:.2f}5802BR5912INVESTFORTE6009CURITIBA62070503{txid}6304E2CA", language="text")
            
            if st.button("Simular Pagamento Confirmado"):
                usr["saldo"] += valor_deposito
                usr["extrato"].append({"tipo": "Depósito Pix", "valor": valor_deposito, "status": "Aprovado", "descricao": f"ID: {txid}"})
                
                # Regra: se depositar >= R$ 200, concede R$ 50 de bónus a quem indicou
                if valor_deposito >= 200.0 and usr.get("indicado_por") and not usr.get("bono_recebido"):
                    quem_indicou = usr["indicado_por"]
                    if quem_indicou in st.session_state.usuarios:
                        st.session_state.usuarios[quem_indicou]["saldo"] += 50.0
                        st.session_state.usuarios[quem_indicou]["extrato"].append({
                            "tipo": "Bónus Indicação",
                            "valor": 50.0,
                            "status": "Aprovado",
                            "descricao": f"Indicação do utilizador: {usr['nome']}"
                        })
                        usr["bono_recebido"] = True
                
                st.success(f"R$ {valor_deposito:.2f} creditados na sua conta!")
                st.rerun()

    # ABA 3: SAQUE VIA PIX
    elif aba_ativa == "📤 Saque Pix":
        st.markdown("### 📤 Solicitar Saque via Pix")
        st.write(f"Saldo disponível: **R$ {usr['saldo']:,.2f}**")

        if st.session_state.saques_bloqueados_global:
            st.error("🔒 **ATENÇÃO:** O sistema de saques e transferências está temporariamente suspenso pela administração.")
        else:
            chave_pix = st.text_input("Sua Chave Pix para Recebimento:", placeholder="CPF, Celular, E-mail ou Aleatória")
            valor_saque = st.number_input("Valor para Saque (R$):", min_value=10.0, max_value=float(usr["saldo"]) if usr["saldo"] >= 10 else 10.0, value=10.0, step=10.0)

            if st.button("Realizar Saque Instantâneo"):
                if not chave_pix.strip():
                    st.error("Informe uma Chave Pix válida.")
                elif valor_saque > usr["saldo"]:
                    st.error("Saldo insuficiente.")
                else:
                    usr["saldo"] -= valor_saque
                    txid_saque = f"SAQ{random.randint(100000, 999999)}"
                    usr["extrato"].append({"tipo": "Saque Pix", "valor": -valor_saque, "status": "Processado", "descricao": f"Chave: {chave_pix} ({txid_saque})"})
                    st.success(f"✅ Saque de R$ {valor_saque:.2f} processado!")
                    st.rerun()

    # ABA 4: INDIQUE E GANHE
    elif aba_ativa == "🤝 Indique & Ganhe":
        st.markdown("### 🎁 Programa de Afiliados")
        st.write("Convide amigos e ganhe bónus na sua conta!")

        st.markdown("""
        <div class="card-indicacao">
            <h4 style="color: #38BDF8; margin-top:0;">Como funciona a regra?</h4>
            <p style="color: #CBD5E1; font-size: 14px;">
            1. Partilhe o seu Código de Indicação exclusivo com os seus amigos.<br>
            2. Quando a pessoa indicada fizer o primeiro depósito de <b>no mínimo R$ 200,00</b>, você recebe automaticamente <b>R$ 50,00 de bónus</b> no seu saldo!
            </p>
        </div>
        """, unsafe_allow_html=True)

        st.text_input("Seu Código de Indicação:", value=usr.get("codigo_ref", "REF123"), disabled=True)

    # ABA 5: EXTRATO INDIVIDUAL
    elif aba_ativa == "📜 Extrato":
        st.markdown("### 📜 Histórico de Movimentações")
        if not usr["extrato"]:
            st.info("Nenhuma movimentação realizada nesta conta.")
        else:
            for t in reversed(usr["extrato"]):
                cor = "#22C55E" if t["valor"] > 0 else "#EF4444"
                st.markdown(f"**{t['tipo']}** — <span style='color:{cor}; font-weight:bold;'>R$ {t['valor']:,.2f}</span> | Status: `{t['status']}`<br><small>{t['descricao']}</small>", unsafe_allow_html=True)
                st.divider()
